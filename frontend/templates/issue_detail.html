{% block scripts %}
const ISSUE_ID = '{{ issue_id }}';
const sevColor = { critical:'red', warning:'amber', info:'blue' };

function fmt(dt) {
  if (!dt) return '—';
  return new Date(dt + (dt.endsWith('Z') ? '' : 'Z')).toLocaleString();
}

async function loadIssue() {
  const r = await fetch(`/issues/${ISSUE_ID}`);
  if (!r.ok) { document.querySelector('.page-title').textContent = 'Issue not found'; return; }
  const i = await r.json();

  document.getElementById('issue-title').textContent = i.title;
  document.getElementById('d-severity').innerHTML  = `${i.severity}`;
  document.getElementById('d-category').textContent = i.category;
  document.getElementById('d-status').innerHTML    = `${i.status}`;
  document.getElementById('d-file').textContent    = i.file_path || '—';
  document.getElementById('d-line').textContent    = i.line_number || '—';
  document.getElementById('d-lang').textContent    = i.language || '—';
  document.getElementById('d-repo').textContent    = i.repo || '—';
  document.getElementById('d-description').textContent = i.description || '—';
  document.getElementById('d-fix').textContent     = i.fix_suggestion || '—';
  document.getElementById('d-followups').textContent = i.followup_count;
  document.getElementById('d-created').textContent  = fmt(i.created_at);
  document.getElementById('d-deadline').textContent  = fmt(i.deadline_at);
  document.getElementById('d-lastfu').textContent    = fmt(i.last_followup_at);
  document.getElementById('d-resolved').textContent  = fmt(i.resolved_at);

  document.getElementById('d-owner').innerHTML = i.owner_name
    ? `

         
${(i.owner_name||'?').split(' ').map(w=>w[0]).join('').substring(0,2).toUpperCase()}

         
${i.owner_name}
${i.owner_email||''}

       
`
    : 'Unassigned';

  document.getElementById('d-jira').innerHTML = i.jira_ticket_url
    ? `${i.jira_ticket_id} ↗`
    : 'Not parked';
  document.getElementById('d-sheet').textContent = i.sheet_row_index ? `Row ${i.sheet_row_index}` : '—';

  // Actions
  const actions = document.getElementById('issue-actions');
  if (i.status !== 'resolved') {
    actions.innerHTML = `
      
      
    `;
  } else {
    actions.innerHTML = `✅ Resolved`;
  }
}

async function setStatus(status) {
  await fetch(`/issues/${ISSUE_ID}`, {
    method:'PATCH',
    headers:{'Content-Type':'application/json'},
    body: JSON.stringify({ status })
  });
  loadIssue();
}

async function reassign() {
  const email = document.getElementById('reassign-email').value.trim();
  const name  = document.getElementById('reassign-name').value.trim();
  if (!email) { alert('Enter an email'); return; }
  await fetch(`/issues/${ISSUE_ID}`, {
    method:'PATCH',
    headers:{'Content-Type':'application/json'},
    body: JSON.stringify({ owner_email: email, owner_name: name || email })
  });
  loadIssue();
}

loadIssue();
